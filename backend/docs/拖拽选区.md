from playwright.sync_api import sync_playwright
import pyperclip, time

with sync_playwright() as p:
    # 必须 headless=False，否则拿不到剪贴板
    browser = p.chromium.launch(headless=False)
    page = browser.new_page()
    page.goto("https://www.zhipin.com/web/geek/resume?...")   # 登录后的简历页

    frame = page.frame_locator('iframe[src*="c-resume"]').first
    canvas = frame.locator('canvas#resume')

    # ---- 1. 等 canvas 渲染完成 ----
    canvas.wait_for_element_state('visible')

    # ---- 2. 计算拖拽起点/终点 ----
    box = canvas.bounding_box()          # 返回 {'x':.., 'y':.., 'width':.., 'height':..}
    sx, sy = box['x'] + 5, box['y'] + 5                 # 左上角往里 5px
    ex, ey = box['x'] + box['width'] - 5, box['y'] + box['height'] - 5   # 右下角往里 5px

    # ---- 3. 拖拽选区 ----
    page.mouse.move(sx, sy)
    page.mouse.down()                    # 按住左键
    page.mouse.move(ex, ey, steps=30)    # 平滑拖到右下角
    page.mouse.up()                      # 松开

    # ---- 4. 复制 ----
    page.keyboard.press('Control+C')
    time.sleep(0.5)                      # 等剪贴板写入完成

    # ---- 5. 取结果 ----
    text = pyperclip.paste()
    print(text)

    browser.close()

    常见踩坑 & 调整
坐标系：bounding_box() 返回的是主页面坐标，所以 mouse.* 直接对 page 用即可。
选区太小：把 +5/-5 去掉或改成 +1/-1 让起点/终点更靠边。
选不上：在 mouse.up() 后加一个 page.keyboard.press('ArrowRight') 再 Ctrl+C，强制触发选区刷新。
剪贴板权限：如果 pyperclip.paste() 为空，改用
text = page.evaluate_handle('navigator.clipboard.readText()').json_value()
