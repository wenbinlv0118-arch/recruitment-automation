```javascript
/**
 * 简历数据验证工具

class ResumeValidator {
  constructor() {
    this.validationRules = {
      basicInfo: {
        name: { required: true, type: 'string', maxLength: 50, minLength: 2 },
        age: { required: false, type: 'number', min: 16, max: 70 },
        gender: { required: false, type: 'string', enum: ['男', '女', '其他'] },
        phone: { required: false, type: 'string', pattern: /^1[3-9]\d{9}$/ },
        email: { required: false, type: 'string', pattern: /^[^\s@]+@[^\s@]+\.[^\s@]+$/ },
        location: { required: false, type: 'string', maxLength: 100 }
      },
      jobIntention: {
        expectedPosition: { required: true, type: 'string', maxLength: 100 },
        expectedSalary: { required: false, type: 'string', maxLength: 50 },
        expectedLocation: { required: false, type: 'string', maxLength: 100 },
        workType: { required: false, type: 'string', enum: ['全职', '兼职', '实习', '远程'] },
        industry: { required: false, type: 'string', maxLength: 100 }
      },
      workExperience: {
        required: false,
        type: 'array',
        items: {
          companyName: { required: true, type: 'string', maxLength: 200 },
          position: { required: true, type: 'string', maxLength: 100 },
          startDate: { required: true, type: 'string', pattern: /^\d{4}-\d{2}$/ },
          endDate: { required: false, type: 'string', pattern: /^(\d{4}-\d{2}|至今)$/ },
          description: { required: false, type: 'string', maxLength: 2000 }
        }
      }
    };
  }

  /**
   * 验证简历数据
   * @param {Object} resumeData - 简历数据
   * @returns {Object} 验证结果
   */
  validate(resumeData) {
    const errors = [];
    const warnings = [];

    try {
      // 验证基本信息
      this.validateSection(resumeData.basicInfo, this.validationRules.basicInfo, 'basicInfo', errors, warnings);
      
      // 验证求职意向
      this.validateSection(resumeData.jobIntention, this.validationRules.jobIntention, 'jobIntention', errors, warnings);
      
      // 验证工作经历
      if (resumeData.workExperience && Array.isArray(resumeData.workExperience)) {
        resumeData.workExperience.forEach((exp, index) => {
          this.validateSection(exp, this.validationRules.workExperience.items, `workExperience[${index}]`, errors, warnings);
        });
      }

      return {
        isValid: errors.length === 0,
        errors,
        warnings,
        score: this.calculateValidationScore(resumeData, errors, warnings)
      };
    } catch (error) {
      return {
        isValid: false,
        errors: [`验证过程出错: ${error.message}`],
        warnings: [],
        score: 0
      };
    }
  }

  /**
   * 验证数据段
   */
  validateSection(data, rules, sectionName, errors, warnings) {
    if (!data && this.hasRequiredFields(rules)) {
      errors.push(`${sectionName} 段落缺失，但包含必填字段`);
      return;
    }

    if (!data) return;

    for (const [field, rule] of Object.entries(rules)) {
      const value = data[field];
      const fieldPath = `${sectionName}.${field}`;

      // 检查必填字段
      if (rule.required && (value === undefined || value === null || value === '')) {
        errors.push(`${fieldPath} 是必填字段`);
        continue;
      }

      // 如果字段为空且非必填，跳过其他验证
      if (value === undefined || value === null || value === '') {
        continue;
      }

      // 类型验证
      if (rule.type && typeof value !== rule.type) {
        errors.push(`${fieldPath} 类型错误，期望 ${rule.type}，实际 ${typeof value}`);
        continue;
      }

      // 字符串长度验证
      if (rule.type === 'string') {
        if (rule.maxLength && value.length > rule.maxLength) {
          errors.push(`${fieldPath} 长度超过限制，最大 ${rule.maxLength} 字符`);
        }
        if (rule.minLength && value.length < rule.minLength) {
          errors.push(`${fieldPath} 长度不足，最小 ${rule.minLength} 字符`);
        }
      }

      // 数字范围验证
      if (rule.type === 'number') {
        if (rule.min !== undefined && value < rule.min) {
          errors.push(`${fieldPath} 值过小，最小值 ${rule.min}`);
        }
        if (rule.max !== undefined && value > rule.max) {
          errors.push(`${fieldPath} 值过大，最大值 ${rule.max}`);
        }
      }

      // 正则表达式验证
      if (rule.pattern && !rule.pattern.test(value)) {
        errors.push(`${fieldPath} 格式不正确`);
      }

      // 枚举值验证
      if (rule.enum && !rule.enum.includes(value)) {
        errors.push(`${fieldPath} 值不在允许范围内: ${rule.enum.join(', ')}`);
      }
    }
  }

  /**
   * 检查规则中是否有必填字段
   */
  hasRequiredFields(rules) {
    return Object.values(rules).some(rule => rule.required);
  }

  /**
   * 计算验证分数
   */
  calculateValidationScore(data, errors, warnings) {
    let score = 100;
    score -= errors.length * 10; // 每个错误扣10分
    score -= warnings.length * 5; // 每个警告扣5分
    return Math.max(0, score);
  }
}

module.exports = new ResumeValidator();
```

```javascript
/**
 * 数据格式转换工具
 */
class DataConverter {
  /**
   * 将旧格式简历数据转换为新格式
   * @param {Object} oldData - 旧格式数据
   * @returns {Object} 新格式数据
   */
  convertOldToNew(oldData) {
    const newData = {
      basicInfo: {
        name: oldData.name || '',
        age: this.parseNumber(oldData.age),
        gender: oldData.gender || null,
        phone: oldData.phone || null,
        email: oldData.email || null,
        location: oldData.location || null,
        workYears: this.parseNumber(oldData.workYears)
      },
      jobIntention: {
        expectedPosition: oldData.expectedPosition?.position || oldData.position || '',
        expectedSalary: oldData.expectedPosition?.salary || oldData.salary || null,
        expectedLocation: oldData.expectedPosition?.location || oldData.location || null,
        workType: oldData.expectedPosition?.workType || null,
        industry: oldData.expectedPosition?.industry || null
      },
      selfEvaluation: oldData.selfIntroduction || oldData.summary || null,
      workExperience: this.convertWorkExperience(oldData.workExperience),
      projectExperience: this.convertProjectExperience(oldData.projectExperience),
      educationBackground: this.convertEducationExperience(oldData.educationExperience),
      skills: this.convertSkills(oldData.skills),
      certificates: oldData.certificates || [],
      languageAbility: oldData.languageAbility || [],
      metadata: {
        parseMethod: oldData.parseMethod || 'traditional',
        parseStatus: oldData.parseStatus || 'completed',
        qualityScore: this.parseNumber(oldData.qualityScore) || 0,
        source: oldData.source || 'unknown',
        originalText: oldData.originalText || '',
        parsedContent: oldData.parsedContent || '',
        parseTime: oldData.parseTime || new Date().toISOString()
      }
    };

    return newData;
  }

  /**
   * 将新格式数据转换为数据库存储格式
   * @param {Object} newData - 新格式数据
   * @returns {Object} 数据库存储格式
   */
  convertToDatabase(newData) {
    const { basicInfo, jobIntention, metadata } = newData;
    
    return {
      // 主表数据
      main: {
        id: this.generateUUID(),
        name: basicInfo.name,
        age: basicInfo.age,
        gender: basicInfo.gender,
        phone: basicInfo.phone,
        email: basicInfo.email,
        location: basicInfo.location,
        work_years: basicInfo.workYears,
        education: this.extractHighestEducation(newData.educationBackground),
        current_status: basicInfo.currentStatus,
        expected_position: jobIntention.expectedPosition,
        expected_salary: jobIntention.expectedSalary,
        expected_location: jobIntention.expectedLocation,
        expected_work_type: jobIntention.workType,
        expected_industry: jobIntention.industry,
        self_evaluation: newData.selfEvaluation,
        raw_content: metadata.originalText,
        parsed_data: JSON.stringify(newData),
        parse_method: metadata.parseMethod,
        parse_status: metadata.parseStatus,
        quality_score: metadata.qualityScore,
        source: metadata.source
      },
      // 关联表数据
      workExperiences: this.convertWorkExperienceForDB(newData.workExperience),
      projectExperiences: this.convertProjectExperienceForDB(newData.projectExperience),
      educationExperiences: this.convertEducationForDB(newData.educationBackground),
      skills: this.convertSkillsForDB(newData.skills)
    };
  }

  /**
   * 转换工作经历
   */
  convertWorkExperience(workExp) {
    if (!Array.isArray(workExp)) return [];
    
    return workExp.map((exp, index) => ({
      companyName: exp.companyName || exp.company || '',
      position: exp.position || '',
      industry: exp.industry || null,
      startDate: this.formatDate(exp.startDate),
      endDate: this.formatDate(exp.endDate),
      isCurrent: exp.isCurrent || exp.endDate === '至今',
      description: exp.description || '',
      achievements: exp.achievements || null,
      technologies: Array.isArray(exp.technologies) ? exp.technologies.join(', ') : exp.technologies,
      sortOrder: index
    }));
  }

  /**
   * 转换项目经历
   */
  convertProjectExperience(projectExp) {
    if (!Array.isArray(projectExp)) return [];
    
    return projectExp.map((proj, index) => ({
      projectName: proj.projectName || proj.name || '',
      role: proj.role || '',
      startDate: this.formatDate(proj.startDate),
      endDate: this.formatDate(proj.endDate),
      description: proj.description || '',
      technologies: Array.isArray(proj.technologies) ? proj.technologies.join(', ') : proj.technologies,
      achievements: proj.achievements || null,
      teamSize: this.parseNumber(proj.teamSize),
      sortOrder: index
    }));
  }

  /**
   * 转换教育经历
   */
  convertEducationExperience(eduExp) {
    if (!Array.isArray(eduExp)) return [];
    
    return eduExp.map((edu, index) => ({
      schoolName: edu.schoolName || edu.school || '',
      major: edu.major || '',
      degree: edu.degree || '',
      startDate: this.formatDate(edu.startDate),
      endDate: this.formatDate(edu.endDate),
      gpa: edu.gpa || null,
      achievements: edu.achievements || null,
      sortOrder: index
    }));
  }

  /**
   * 转换技能信息
   */
  convertSkills(skills) {
    if (!Array.isArray(skills)) return [];
    
    return skills.map(skill => {
      if (typeof skill === 'string') {
        return {
          skillName: skill,
          skillCategory: this.categorizeSkill(skill),
          proficiencyLevel: null,
          yearsOfExperience: null
        };
      }
      
      return {
        skillName: skill.name || skill.skillName || '',
        skillCategory: skill.category || this.categorizeSkill(skill.name),
        proficiencyLevel: skill.level || skill.proficiencyLevel || null,
        yearsOfExperience: this.parseNumber(skill.years) || null
      };
    });
  }

  /**
   * 工具方法
   */
  parseNumber(value) {
    if (typeof value === 'number') return value;
    if (typeof value === 'string') {
      const num = parseInt(value.replace(/[^\d]/g, ''));
      return isNaN(num) ? null : num;
    }
    return null;
  }

  formatDate(dateStr) {
    if (!dateStr) return null;
    if (dateStr === '至今' || dateStr === 'present') return '至今';
    
    // 尝试解析各种日期格式
    const patterns = [
      /^(\d{4})-(\d{2})$/, // 2023-01
      /^(\d{4})\.(\d{2})$/, // 2023.01
      /^(\d{4})年(\d{1,2})月$/, // 2023年1月
      /^(\d{4})$/ // 2023
    ];
    
    for (const pattern of patterns) {
      const match = dateStr.match(pattern);
      if (match) {
        const year = match[1];
        const month = match[2] ? match[2].padStart(2, '0') : '01';
        return `${year}-${month}`;
      }
    }
    
    return dateStr; // 返回原始值
  }

  categorizeSkill(skillName) {
    const categories = {
      '编程语言': ['Java', 'Python', 'JavaScript', 'C++', 'C#', 'Go', 'Rust', 'Swift', 'Kotlin'],
      '前端技术': ['React', 'Vue', 'Angular', 'HTML', 'CSS', 'TypeScript', 'jQuery'],
      '后端技术': ['Spring', 'Django', 'Express', 'Flask', 'Laravel', 'Rails'],
      '数据库': ['MySQL', 'PostgreSQL', 'MongoDB', 'Redis', 'Oracle', 'SQLServer'],
      '云服务': ['AWS', 'Azure', '阿里云', '腾讯云', 'Docker', 'Kubernetes'],
      '工具软件': ['Git', 'Jenkins', 'JIRA', 'Confluence', 'Postman']
    };
    
    for (const [category, skills] of Object.entries(categories)) {
      if (skills.some(skill => skillName.toLowerCase().includes(skill.toLowerCase()))) {
        return category;
      }
    }
    
    return '其他';
  }

  extractHighestEducation(educationList) {
    if (!Array.isArray(educationList) || educationList.length === 0) return null;
    
    const degreeOrder = ['博士', '硕士', '本科', '专科', '高中', '初中'];
    
    for (const degree of degreeOrder) {
      if (educationList.some(edu => edu.degree && edu.degree.includes(degree))) {
        return degree;
      }
    }
    
    return educationList[0].degree || null;
  }

  generateUUID() {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
      const r = Math.random() * 16 | 0;
      const v = c === 'x' ? r : (r & 0x3 | 0x8);
      return v.toString(16);
    });
  }

  // 数据库存储格式转换方法
  convertWorkExperienceForDB(workExp) {
    return workExp.map(exp => ({
      id: this.generateUUID(),
      company_name: exp.companyName,
      position: exp.position,
      industry: exp.industry,
      start_date: exp.startDate,
      end_date: exp.endDate,
      is_current: exp.isCurrent,
      description: exp.description,
      achievements: exp.achievements,
      technologies: exp.technologies,
      sort_order: exp.sortOrder
    }));
  }

  convertProjectExperienceForDB(projectExp) {
    return projectExp.map(proj => ({
      id: this.generateUUID(),
      project_name: proj.projectName,
      role: proj.role,
      start_date: proj.startDate,
      end_date: proj.endDate,
      description: proj.description,
      technologies: proj.technologies,
      achievements: proj.achievements,
      team_size: proj.teamSize,
      sort_order: proj.sortOrder
    }));
  }

  convertEducationForDB(eduExp) {
    return eduExp.map(edu => ({
      id: this.generateUUID(),
      school_name: edu.schoolName,
      major: edu.major,
      degree: edu.degree,
      start_date: edu.startDate,
      end_date: edu.endDate,
      gpa: edu.gpa,
      achievements: edu.achievements,
      sort_order: edu.sortOrder
    }));
  }

  convertSkillsForDB(skills) {
    return skills.map(skill => ({
      id: this.generateUUID(),
      skill_name: skill.skillName,
      skill_category: skill.skillCategory,
      proficiency_level: skill.proficiencyLevel,
      years_of_experience: skill.yearsOfExperience
    }));
  }
}
```

```javascript
const DataConverter = require('../utils/dataConverter');
const ResumeValidator = require('../utils/resumeValidator');

class ResumeParserService {
  // ... 保留现有的构造函数和基础方法 ...

  /**
   * 解析文本简历（重构版本）
   * @param {string} text - 简历文本内容
   * @returns {Object} 标准化格式的解析结果
   */
  parseResumeText(text) {
    if (!text || typeof text !== 'string') {
      throw new Error('无效的文本内容');
    }

    // 使用原有解析逻辑获取基础数据
    const rawResult = this.parseResumeTextLegacy(text);
    
    // 转换为标准化格式
    const standardizedResult = DataConverter.convertOldToNew(rawResult);
    
    // 验证数据
    const validation = ResumeValidator.validate(standardizedResult);
    
    // 添加验证结果到元数据
    standardizedResult.metadata.validationResult = validation;
    standardizedResult.metadata.completenessScore = this.calculateCompletenessScore(standardizedResult);
    standardizedResult.metadata.richnessScore = this.calculateRichnessScore(standardizedResult);
    
    return standardizedResult;
  }

  /**
   * 原有解析逻辑（重命名）
   */
  parseResumeTextLegacy(text) {
    // 保留原有的解析逻辑
    const result = {
      name: this.extractName(text),
      age: this.extractAge(text),
      workYears: this.extractWorkYears(text),
      education: this.extractEducation(text),
      currentStatus: this.extractCurrentStatus(text),
      phone: this.extractPhone(text),
      email: this.extractEmail(text),
      selfIntroduction: this.extractSelfIntroduction(text),
      expectedPosition: this.extractExpectedPosition(text),
      positionExperience: this.extractPositionExperience(text),
      workExperience: this.extractWorkExperience(text),
      educationExperience: this.extractEducationExperience(text),
      certificates: this.extractCertificates(text),
      volunteerExperience: this.extractVolunteerExperience(text),
      skills: this.extractSkills(text),
      parseStatus: 'completed',
      qualityScore: this.calculateQualityScore(text),
      originalText: text
    };

    return result;
  }

  /**
   * 计算完整性分数
   */
  calculateCompletenessScore(resumeData) {
    const requiredFields = [
      'basicInfo.name',
      'jobIntention.expectedPosition'
    ];
    
    const optionalFields = [
      'basicInfo.age',
      'basicInfo.phone',
      'basicInfo.email',
      'selfEvaluation',
      'workExperience',
      'educationBackground'
    ];
    
    let score = 0;
    let totalWeight = 0;
    
    // 必填字段权重更高
    requiredFields.forEach(field => {
      totalWeight += 30;
      if (this.getNestedValue(resumeData, field)) {
        score += 30;
      }
    });
    
    // 可选字段
    optionalFields.forEach(field => {
      totalWeight += 10;
      const value = this.getNestedValue(resumeData, field);
      if (value && (Array.isArray(value) ? value.length > 0 : true)) {
        score += 10;
      }
    });
    
    return Math.round((score / totalWeight) * 100);
  }

  /**
   * 计算丰富度分数
   */
  calculateRichnessScore(resumeData) {
    let score = 0;
    
    // 工作经历丰富度
    if (resumeData.workExperience && resumeData.workExperience.length > 0) {
      score += Math.min(resumeData.workExperience.length * 10, 30);
      
      // 检查工作经历详细程度
      const detailedExperiences = resumeData.workExperience.filter(exp => 
        exp.description && exp.description.length > 50
      );
      score += detailedExperiences.length * 5;
    }
    
    // 项目经历丰富度
    if (resumeData.projectExperience && resumeData.projectExperience.length > 0) {
      score += Math.min(resumeData.projectExperience.length * 8, 20);
    }
    
    // 技能丰富度
    if (resumeData.skills && resumeData.skills.length > 0) {
      score += Math.min(resumeData.skills.length * 2, 20);
    }
    
    // 教育背景
    if (resumeData.educationBackground && resumeData.educationBackground.length > 0) {
      score += 15;
    }
    
    // 自我评价
    if (resumeData.selfEvaluation && resumeData.selfEvaluation.length > 50) {
      score += 15;
    }
    
    return Math.min(score, 100);
  }

  /**
   * 获取嵌套对象的值
   */
  getNestedValue(obj, path) {
    return path.split('.').reduce((current, key) => {
      return current && current[key] !== undefined ? current[key] : null;
    }, obj);
  }

  // ... 保留其他现有方法 ...
}
```
